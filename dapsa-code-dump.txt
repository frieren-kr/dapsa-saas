"use server";

import { redirect } from "next/navigation";
import prisma from "@/lib/prisma";
import { requireAuth } from "@/lib/session";
import { createProjectSchema } from "@/lib/validations";
import { revalidatePath } from "next/cache";
import { isProjectOrganizer } from "@/lib/permissions";
import { updateProjectSchema } from "@/lib/validations";

export async function createProject(
  _prevState: { error?: string } | null,
  formData: FormData
) {
  // 1. ?몄쬆 ?뺤씤
  const session = await requireAuth();

  // 2. 沅뚰븳 ?뺤씤 - ORGANIZER留??앹꽦 媛??
  if (session.user.role !== "ORGANIZER") {
    return { error: "?듭궗 以鍮꾧린愿留??꾨줈?앺듃瑜?留뚮뱾 ???덉뼱?? };
  }

  // 3. ?낅젰媛??뚯떛 諛?寃利?
  const raw = {
    title: formData.get("title") as string,
    description: (formData.get("description") as string) || undefined,
    startDate: (formData.get("startDate") as string) || undefined,
    endDate: (formData.get("endDate") as string) || undefined,
  };

  const parsed = createProjectSchema.safeParse(raw);
  if (!parsed.success) {
    // 泥?踰덉㎏ ?먮윭 硫붿떆吏 諛섑솚
    return { error: parsed.error.issues[0].message };
  }

  const { title, description, startDate, endDate } = parsed.data;

  // 4. DB?????
  const project = await prisma.project.create({
    data: {
      title,
      description,
      startDate: startDate ? new Date(startDate) : null,
      endDate: endDate ? new Date(endDate) : null,
      organizerId: session.user.id,
      status: "DRAFT",
    },
  });

  // 5. ??쒕낫?쒕줈 由щ떎?대젆??(?꾨줈?앺듃 紐⑸줉? 4-3?먯꽌 留뚮뱾 ?덉젙)
  redirect(`/dashboard?created=${project.id}`);
}


export async function updateProject(input: {
  projectId: string;
  title: string;
  description?: string;
  startDate?: string;
  endDate?: string;
}) {
  const session = await requireAuth();

  const parsed = updateProjectSchema.safeParse(input);
  if (!parsed.success) {
    return { error: parsed.error.issues[0].message };
  }

  // 沅뚰븳 寃??- ???꾨줈?앺듃??organizer留??섏젙 媛??
  const isOwner = await isProjectOrganizer(session.user.id, input.projectId);
  if (!isOwner) {
    return { error: "?섏젙 沅뚰븳???놁뼱?? };
  }

  await prisma.project.update({
    where: { id: input.projectId },
    data: {
      title: input.title,
      description: input.description || null,
      startDate: input.startDate ? new Date(input.startDate) : null,
      endDate: input.endDate ? new Date(input.endDate) : null,
    },
  });

  revalidatePath(`/projects/${input.projectId}`);
  revalidatePath("/dashboard");
  return { success: true };
}

export async function deleteProject(input: { projectId: string }) {
  const session = await requireAuth();

  // 沅뚰븳 寃??
  const isOwner = await isProjectOrganizer(session.user.id, input.projectId);
  if (!isOwner) {
    return { error: "??젣 沅뚰븳???놁뼱?? };
  }

  // ??젣 - Cascade ?ㅼ젙 ?뚮Ц???섏쐞 ?곗씠??sites, schedules, invitations, members)???먮룞 ??젣
  await prisma.project.delete({
    where: { id: input.projectId },
  });

  revalidatePath("/dashboard");
  return { success: true };
}

function toDateKey(date: Date): string {
  const d = new Date(date);
  const yyyy = d.getFullYear();
  const mm = String(d.getMonth() + 1).padStart(2, "0");
  const dd = String(d.getDate()).padStart(2, "0");

}

// ?좎쭨蹂?洹몃９??+ 媛??좎쭨 ?덉뿉???쒓컙???뺣젹(諛⑹뼱??
function groupByDate(schedules: Schedule[]): Record<string, Schedule[]> {
  const result: Record<string, Schedule[]> = {};
  for (const s of schedules) {
    const key = toDateKey(s.date);
    if (!result[key]) result[key] = [];
    result[key].push(s);
  }
  for (const key of Object.keys(result)) {
    result[key].sort((a, b) => a.startTime.localeCompare(b.startTime));
  }
  return result;
}

interface RoutePoint {
  latitude: number;
  longitude: number;
}

interface RouteLeg {
  distance: number; // 援ш컙 嫄곕━-誘명꽣
  duration: number; //援ш컙 ?쒓컙-諛由ъ큹
}

interface RouteResult {
  path: number[][];      // [[lng, lat], ...] 寃쎈줈 醫뚰몴
  distance: number;      // 誘명꽣
  duration: number;      // 諛由ъ큹
  legs: RouteLeg[];        // 援ш컙蹂?嫄곕━/?쒓컙
  usedPointCount: number; //?ㅼ젣 寃쎈줈???ъ슜??吏?????뺣났 ?쒖쇅)
}


export async function calculateRoute(
  points: RoutePoint[]
): Promise<RouteResult | null> {
  // ?듭궗吏媛 2媛?誘몃쭔?대㈃ 寃쎈줈 ?놁쓬
  if (points.length < 2) {
    return null;
  }

  const clientId = process.env.NEXT_PUBLIC_NAVER_MAP_CLIENT_ID;
  const clientSecret = process.env.NAVER_MAP_CLIENT_SECRET;

  if (!clientId || !clientSecret) {
    throw new Error("?ㅼ씠踰?吏??API ?ㅺ? ?ㅼ젙?섏? ?딆븯?댁슂");
  }
  //?뺣났 媛먯튂??泥?吏?먭낵 留덉?留?吏?먯씠 媛숈쑝硫?留덉?留??쒖쇅
  let routePoints = points;
  if (points.length >= 3) {
    const first =points[0];
    const last = points[points.length -1];
    if (
      first.latitude === last.latitude &&
      first.longitude === last.longitude
    ) {
      routePoints = points.slice(0,-1); // 留덉?留?吏???쒖쇅
    }
  }
  
  const start = routePoints[0];
  const goal = routePoints[routePoints.length - 1];
  const waypoints = routePoints.slice(1, -1); // 以묎컙 ?듭궗吏??

  // 醫뚰몴??"寃쎈룄,?꾨룄" ?뺤떇 (?ㅼ씠踰?洹쒖튃: 寃쎈룄 癒쇱?)
  const startParam = `${start.longitude},${start.latitude}`;
  const goalParam = `${goal.longitude},${goal.latitude}`;

  // 寃쎌쑀吏??肄쒕줎(:)?쇰줈 援щ텇
  const waypointsParam = waypoints
    .map((w) => `${w.longitude},${w.latitude}`)
    .join(":");

  // API URL 援ъ꽦
  const url = new URL(
    "https://maps.apigw.ntruss.com/map-direction-15/v1/driving"
  );
  url.searchParams.set("start", startParam);
  url.searchParams.set("goal", goalParam);
  if (waypointsParam) {
    url.searchParams.set("waypoints", waypointsParam);
  }
  url.searchParams.set("option", "trafast"); // ?ㅼ떆媛?鍮좊Ⅸ湲? 寃쎌쑀吏 ?쒖꽌 ?좎?

  // API ?몄텧 - ?ㅻ뜑???몄쬆 ?뺣낫
  const response = await fetch(url.toString(), {
    headers: {
      "x-ncp-apigw-api-key-id": clientId,
      "x-ncp-apigw-api-key": clientSecret,
    },
  });

  if (!response.ok) {
    throw new Error(`寃쎈줈 怨꾩궛 ?ㅽ뙣: ${response.status}`);
  }


  // ?묐떟 肄붾뱶 ?뺤씤 (0???깃났)
  const data = await response.json();
  if (data.code !== 0) {
    throw new Error(`寃쎈줈瑜?李얠쓣 ???놁뼱?? ${data.message || "?????녿뒗 ?ㅻ쪟"}`);
  }
  
  // 寃곌낵 ?뚯떛 - trafast(鍮좊Ⅸ 寃쎈줈) ?ъ슜
  const route = data.route?.trafast?.[0];
  if (!route) {
    throw new Error("寃쎈줈 ?곗씠?곌? ?놁뼱??);
  }
  //援ш컙蹂??뺣낫 議곕┰
  //waypoints諛곗뿴 : 媛?寃쎌쑀吏源뚯???援ш컙?뺣낫
  //goal: 留덉?留?寃쎌쑀吏 -> 紐⑹쟻吏 援ш컙 ?뺣낫
  const summary = route.summary;
  const legs: RouteLeg[] = [];

  //寃쎌쑀吏?ㅼ쓽 援ш컙
  if (summary.waypoints && Array.isArray(summary.waypoints)) {
    for (const wp of summary.waypoints) {
      legs.push({
        distance: wp.distance ?? 0,
        duration: wp.duration ?? 0,
      });
    }
  }

  //留덉?留?紐⑹쟻吏 援ш컙
  legs.push({
    distance: summary.goal.distance ?? 0,
    duration: summary.goal.duration ?? 0,
  });

  return {
    path: route.path,                    // ?꾨줈 醫뚰몴 諛곗뿴
    distance: route.summary.distance,    // 誘명꽣
    duration: route.summary.duration,    // 諛由ъ큹
    legs,
    usedPointCount: routePoints.length, //?뺣났 ?쒖쇅 ?ㅼ젣 吏????
  };
}

export interface Coords{
    latitude: number;
    longitude: number;
    accuracy: number;
}

export function getCurrentPosition(): Promise<Coords> {
    return new Promise((resolve, reject) => {
        if (!navigator.geolocation) {
            reject(new Error("??釉뚮씪?곗????꾩튂 湲곕뒫??吏?먰븯吏 ?딆븘??));
            return;
        }

        navigator.geolocation.getCurrentPosition(
            (position) =>{
                resolve({
                    latitude: position.coords.latitude,
                    longitude: position.coords.longitude,
                    accuracy: position.coords.accuracy,
                });
            },
            (error) =>{
                if (error.code === error.PERMISSION_DENIED) {
                    reject(new Error("?꾩튂 沅뚰븳??嫄곕??먯뼱?? 釉뚮씪?곗? ?ㅼ젙?먯꽌 ?덉슜?댁＜?몄슂"));
                } else if (error.code === error.POSITION_UNAVAILABLE) {
                    reject(new Error("?꾩튂瑜??뺤씤?????놁뼱??));
                } else if (error.code === error.TIMEOUT) {
                    reject(new Error("?꾩튂 ?뺤씤 ?쒓컙??珥덇낵?먯뼱??));
                } else {
                    reject(new Error("?꾩튂瑜?媛?몄삤?붾뜲 ?ㅽ뙣?덉뼱??));
                }
            },
            {enableHighAccuracy: true,
                timeout: 10000,
                maximumAge: 0,
            }
        );
    });
}
